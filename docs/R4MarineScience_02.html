## 8. Statistical summary
# Calculate summary statistics by species and zone
summary_table <- master_data |>
  group_by(scientific_name, zone) |>
  summarise(
    mean_catch = mean(count, na.rm = TRUE),
    se_catch = sd(count, na.rm = TRUE) /
      sqrt(sum(!is.na(count))),
    
    avg_salinity = mean(mean_salinity, na.rm = TRUE),
    se_salinity = sd(mean_salinity, na.rm = TRUE) /
      sqrt(sum(!is.na(mean_salinity))),
    
    .groups = "drop") |>
  rename(
    mean_salinity = avg_salinity)
